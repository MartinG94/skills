\cvsection{Actividades y Vinculación Institucional}

\begin{cvitems}
    \item \textbf{Liderazgo Social y Ambiental (AVEIT):} Conducción de proyectos solidarios con impacto en merenderos de Córdoba y campañas de recolección de residuos electrónicos (RAEE) para reciclaje sustentable.
    \item \textbf{Modernización de Procesos Institucionales:} Digitalización de registros asamblearios y trazabilidad de proyectos comunitarios mediante tableros colaborativos y herramientas ágiles.
    \item \textbf{Participación Asamblearia y Formativa:} Representación estudiantil en órganos de gobierno universitario y jornadas académicas de ingeniería (UTN-FRC).
\end{cvitems}
